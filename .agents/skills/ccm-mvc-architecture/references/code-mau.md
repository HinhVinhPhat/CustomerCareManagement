# Code samples (ASP.NET Core MVC + EF Core 8 or later)

Only the parts that are easy to get wrong. These have not been run on a real machine yet: build,
run and try them; the entity names generated by the scaffold may differ from the example (check
`Models/Entities` after scaffolding).

Contents: Program.cs | Configure trigger tables | SESSION_CONTEXT interceptor | Translate SQL
errors | Service calling a procedure | Service reading a procedure/view | Background work

## Configure trigger tables (Models/EF/ApplicationDbContext.Custom.cs)

```csharp
using CcmWeb.Models.Entities;
using Microsoft.EntityFrameworkCore;

namespace CcmWeb.Models.EF;

public partial class ApplicationDbContext
{
    // The scaffold generates the call site for this method, we only implement it in a separate
    // file so re-scaffolding does not lose it.
    partial void OnModelCreatingPartial(ModelBuilder modelBuilder)
    {
        // Trigger tables: turn off OUTPUT in EF's INSERT/UPDATE (otherwise SQL error 334 on SaveChanges).
        modelBuilder.Entity<Contract>().ToTable(tb => tb.UseSqlOutputClause(false));
        modelBuilder.Entity<ContractDetail>().ToTable(tb => tb.UseSqlOutputClause(false));
        modelBuilder.Entity<Customer>().ToTable(tb => tb.UseSqlOutputClause(false));
        modelBuilder.Entity<Feedback>().ToTable(tb => tb.UseSqlOutputClause(false));
    }
}
```

## Interceptor that sets SESSION_CONTEXT (Helpers or Models/EF)

```csharp
using System.Data.Common;
using Microsoft.EntityFrameworkCore.Diagnostics;

public sealed class SessionContextInterceptor(IHttpContextAccessor accessor) : DbConnectionInterceptor
{
    public override async Task ConnectionOpenedAsync(DbConnection connection,
        ConnectionEndEventData eventData, CancellationToken cancellationToken = default)
        => await SetActorAsync(connection, cancellationToken);

    public override void ConnectionOpened(DbConnection connection, ConnectionEndEventData eventData)
        => SetActorAsync(connection, default).GetAwaiter().GetResult();

    private async Task SetActorAsync(DbConnection connection, CancellationToken ct)
    {
        var employeeID = accessor.HttpContext?.User.GetEmployeeID();      // extension in Helpers/Utils.cs
        if (employeeID is null) return;                              // background work: no acting user
        await using var cmd = connection.CreateCommand();
        cmd.CommandText = "EXEC sp_set_session_context N'EmployeeID', @employeeID";
        var p = cmd.CreateParameter();
        p.ParameterName = "@employeeID";
        p.Value = employeeID.Value;
        cmd.Parameters.Add(p);
        await cmd.ExecuteNonQueryAsync(ct);
    }
}
```

```csharp
// Helpers/Utils.cs
public static int? GetEmployeeID(this ClaimsPrincipal user) =>
    int.TryParse(user.FindFirst("employeeID")?.Value, out var n) ? n : null;
```

## Translating SQL errors (Helpers/Utils.cs) and the business exception

```csharp
public sealed class BusinessException(string message, Exception? inner = null) : Exception(message, inner);

public static string? TranslateSqlError(Exception ex)
{
    for (var e = ex; e is not null; e = e.InnerException)
    {
        if (e is not SqlException s) continue;             // Microsoft.Data.SqlClient.SqlException
        return s.Number switch
        {
            >= 50001 and <= 50015 => s.Message,            // DB THROW: already English
            547 => TranslateConstraintError(s.Message),
            2601 or 2627 => TranslateDuplicateError(s.Message),
            _ => null                                      // other errors: log on the server, show a generic message
        };
    }
    return null;
}

static string TranslateConstraintError(string m) =>
    m.Contains("CHK_Feedback_Assignee") ? "Assign an employee before moving the feedback to a processing status." :
    m.Contains("CHK_Customer_Business")   ? "A business customer cannot have a gender or date of birth." :
    m.Contains("CHK_Customer_Contact")       ? "Enter at least an email or a phone number." :
    m.Contains("CHK_Care_SatisfactionScore")          ? "The 1-5 satisfaction score is only used for the Survey care type." :
    m.Contains("CHK_Care_NextAppointment")           ? "The next appointment date cannot be before the care date." :
    m.Contains("FOREIGN KEY")          ? "This record is still referenced elsewhere." :
                                         "Invalid data.";

static string TranslateDuplicateError(string m) =>
    m.Contains("UX_Customer_Email")          ? "This email already belongs to another customer." :
    m.Contains("UX_Employee_DepartmentManager")    ? "This department already has a manager." :
                                         "Duplicate data (email, phone, name...).";
```

Used in a Service:

```csharp
try { ... }
catch (Exception ex) when (Utils.TranslateSqlError(ex) is { } message)
{
    throw new BusinessException(message, ex);
}
```

A SQL error not in the table (`TranslateSqlError` returns `null`) is not caught here; let the
global error middleware log it and show the error page, without exposing SQL details.

## Service calling a procedure with OUTPUT (ContractService)

```csharp
public async Task<int> AddAsync(int currentEmployeeID, ContractForm f)
{
    // The JSON property names must match case exactly: ProductID, Quantity, UnitPrice
    var json = JsonSerializer.Serialize(
        f.Items.Select(i => new { ProductID = i.ProductID, Quantity = i.Quantity, UnitPrice = i.UnitPrice }),
        new JsonSerializerOptions { PropertyNamingPolicy = null });

    var contractID = new SqlParameter("@ContractID", SqlDbType.Int) { Direction = ParameterDirection.Output };
    try
    {
        await _db.Database.ExecuteSqlRawAsync(
            "EXEC dbo.sp_AddContract @CustomerID=@CustomerID, @EmployeeID=@EmployeeID, @EndDate=@EndDate, @DetailJson=@DetailJson, " +
            "@SignDate=@SignDate, @Note=@Note, @Status=@Status, @ContractID=@ContractID OUTPUT",
            new SqlParameter("@CustomerID", f.CustomerID),
            new SqlParameter("@EmployeeID", f.EmployeeID ?? currentEmployeeID),
            new SqlParameter("@EndDate", SqlDbType.Date) { Value = f.EndDate.ToDateTime(TimeOnly.MinValue) },
            new SqlParameter("@DetailJson", json),
            new SqlParameter("@SignDate", SqlDbType.Date) { Value = f.SignDate is null ? DBNull.Value : f.SignDate.Value.ToDateTime(TimeOnly.MinValue) },
            new SqlParameter("@Note", (object?)f.Note ?? DBNull.Value),
            new SqlParameter("@Status", f.Status ?? WorkConstants.Contract_Active),
            contractID);
    }
    catch (Exception ex) when (Utils.TranslateSqlError(ex) is { } message) { throw new BusinessException(message, ex); }
    return (int)contractID.Value;
}
```

Procedures without OUTPUT (sign, renew, cancel, assign, change status) are also called with
`ExecuteSqlRawAsync` using `EXEC dbo.sp_...` in the same way, wrapped in the same `catch`.

## Service reading a procedure and a view

```csharp
// ViewModels/CustomerRow.cs: property names match the procedure's returned column names exactly
// (including OwnerName)
public class CustomerRow
{
    public int CustomerID { get; set; }
    public string CustomerName { get; set; } = "";
    public string CustomerType { get; set; } = "";
    public string? Email { get; set; }
    public string? Phone { get; set; }
    public string? Address { get; set; }
    public string Tier { get; set; } = "";
    public DateOnly CreatedDate { get; set; }
    public int? OwnerEmployeeID { get; set; }
    public string? OwnerName { get; set; }
    public bool IsActive { get; set; }
    public decimal TotalValue { get; set; }
}

public Task<List<CustomerRow>> SearchAsync(string? keyword, string? tier, int? ownerEmployeeID, bool activeOnly) =>
    _db.Database.SqlQueryRaw<CustomerRow>(
        "EXEC dbo.sp_SearchCustomer @Keyword=@Keyword, @Tier=@Tier, @OwnerEmployeeID=@OwnerEmployeeID, @ActiveOnly=@ActiveOnly",
        new SqlParameter("@Keyword", (object?)keyword ?? DBNull.Value),
        new SqlParameter("@Tier", (object?)tier ?? DBNull.Value),
        new SqlParameter("@OwnerEmployeeID", (object?)ownerEmployeeID ?? DBNull.Value),
        new SqlParameter("@ActiveOnly", activeOnly)).ToListAsync();

// Reading a view (keyless entity generated by the scaffold, usually named VwExpiringContract)
public Task<List<VwExpiringContract>> ExpiringAsync() =>
    _db.VwExpiringContracts.AsNoTracking().OrderBy(v => v.EndDate).ToListAsync();
```

A procedure that returns a single row (like `sp_DashboardSummary`) also uses `SqlQueryRaw<T>` with
`T` a class whose properties match the column names, then `.ToListAsync()` and take the first
element. With a simple type like `int` the column must be aliased `Value`, so prefer a result
class with the real column names as in the example above.

## Background work (the DB has no SQL Agent)

```csharp
public class UpdateResult { public int UpdatedContracts { get; set; } }

public sealed class ContractExpiryService(IServiceScopeFactory scopes, ILogger<ContractExpiryService> log) : BackgroundService
{
    protected override async Task ExecuteAsync(CancellationToken ct)
    {
        using var timer = new PeriodicTimer(TimeSpan.FromHours(24));
        do
        {
            try
            {
                using var scope = scopes.CreateScope();
                var db = scope.ServiceProvider.GetRequiredService<ApplicationDbContext>();
                var result = await db.Database.SqlQueryRaw<UpdateResult>("EXEC dbo.sp_ExpireContracts").ToListAsync(ct);
                log.LogInformation("Moved {Count} contracts to Expired", result.FirstOrDefault()?.UpdatedContracts ?? 0);
            }
            catch (Exception ex) { log.LogError(ex, "Failed to expire overdue contracts"); }
        } while (await timer.WaitForNextTickAsync(ct));
    }
}
```
